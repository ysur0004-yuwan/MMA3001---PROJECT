 35def load_class_names(dataset_path: Path) -> list[str]:
 36    """Loads the class ID -> class name mapping from the dataset's data.yaml.
 37
 38    Args:
 39        dataset_path: Root folder of the downloaded Roboflow dataset
 40            (the folder containing data.yaml, train/, valid/, test/).
 41
 42    Returns:
 43        A list of class names, ordered by class ID (index 0 = class ID 0).
 44
 45    Raises:

 58def parse_yolo_label_file(label_path: Path) -> list[int]:
 59    """Parses a single YOLO-format .txt label file into a list of class IDs.
 60
 61    Each non-empty line in a YOLO label file has the format:
 62        "<class_id> <x_center> <y_center> <width> <height>"
 63    Only the class_id is needed to build image-level labels; the bounding
 64    box coordinates are not used, since this project performs image-level
 65    classification rather than object detection (see Section 1 of the
 66    project documentation for the rationale behind that scope decision).
 67
 68    Args:
 69        label_path: Path to a single .txt annotation file.
 70
 71    Returns:
 72        A list of integer class IDs, one per annotated bounding box in the
 73        image. An empty list means the image has no annotated defects
 74        (a "pass" image).
 75    """
 76    if not label_path.exists():
 77        # No label file at all is treated the same as an empty one: a pass image.
 78        # This matches how Roboflow exports images with zero annotations.
 79        return []
 80
 81    with open(label_path) as f:
 82        lines = [line.strip() for line in f if line.strip()]
 83
 84    class_ids = []
 85    for line in lines:
 86        try:
 87            class_ids.append(int(line.split()[0]))
 88        except (ValueError, IndexError):
 89            # Malformed line (e.g. truncated write, corrupt export) — log and
 90            # skip rather than crashing the whole data audit over one line.
 91            logger.warning(f"Malformed annotation line skipped in {label_path}: '{line}'")
 92    return class_ids

 95def is_image_readable(image_path: Path) -> bool:
 96    """Checks whether an image file can actually be opened and decoded.
 97
 98    Used to filter out corrupt or unreadable files during data loading, per
 99    the project's documented handling of invalid inputs (Section 2 of the
100    project documentation).
101
102    Args:
103        image_path: Path to the image file to check.
104
105    Returns:
106        True if the image opens and verifies successfully, False otherwise
107        (a warning is logged in the False case).
108    """
109    try:
110        with Image.open(image_path) as img:
111            img.verify()
112        return True
113    except (UnidentifiedImageError, OSError) as e:
114        logger.warning(f"Unreadable image skipped: {image_path} ({e})")
115        return False

118def build_image_level_labels(dataset_path: Path, class_names: list[str]) -> pd.DataFrame:
119    """Builds a DataFrame of image-level labels for every image in every split.
120
121    For each image, this function:
122      1. Parses its YOLO label file to get the set of annotated defect classes.
123      2. Derives a pass/fail label: "fail" if one or more defects are
124         annotated, "pass" otherwise.
125      3. Derives an extension (defect-type) label, but ONLY for images with
126         exactly one distinct defect class. Images with zero defects (pass)
127         or more than one distinct defect class (multi-defect) get
128         extension_label = None, since forcing a single label onto a
129         genuinely multi-defect image would misrepresent it — see the
130         project documentation (Section 2) for the policy rationale.
131      4. Skips images that fail a basic readability check.
132
133    Args:
134        dataset_path: Root folder of the downloaded Roboflow dataset.
135        class_names: Class ID -> name mapping, from load_class_names().
136
137    Returns:
138        A pandas DataFrame with one row per valid image, and columns:
139        "image_path", "split", "pass_fail", "is_multi_defect",
140        "extension_label".
141    """
142    records = []
143    for split in ["train", "valid", "test"]:
144        image_dir = dataset_path / split / "images"
145        label_dir = dataset_path / split / "labels"
146
147        if not image_dir.exists():
148            logger.warning(f"Split '{split}' not found at {image_dir}, skipping.")
149            continue
150
151        image_paths = sorted(image_dir.glob("*.jpg")) + sorted(image_dir.glob("*.png"))
152
153        for image_path in image_paths:
154            if not is_image_readable(image_path):
155                continue  # corrupt/unreadable images are excluded entirely
156
157            label_path = label_dir / f"{image_path.stem}.txt"
158            class_ids = parse_yolo_label_file(label_path)
159            unique_classes = sorted(set(class_ids))
160
161            pass_fail = "fail" if len(class_ids) > 0 else "pass"
162            is_multi_defect = len(unique_classes) > 1
163
164            if pass_fail == "pass":
165                extension_label = None
166            elif is_multi_defect:
167                extension_label = None  # excluded from extension training, per policy
168            else:
169                extension_label = class_names[unique_classes[0]]
170
171            records.append({
172                "image_path": str(image_path),
173                "split": split,
174                "pass_fail": pass_fail,
175                "is_multi_defect": is_multi_defect,
176                "extension_label": extension_label,
177            })
178
179    df = pd.DataFrame.from_records(records)
180    logger.info(f"Built image-level labels for {len(df)} images across all splits.")
181    return df

207def _load_and_preprocess_image(path: tf.Tensor, label: tf.Tensor) -> tuple[tf.Tensor, tf.Tensor]:
208    """Reads a single image file off disk and prepares it for the model.
209
210    Shared by both make_pass_fail_dataset() and make_defect_type_dataset(),
211    since both tasks use identical image preprocessing — only the label
212    meaning differs between the two tasks, not how the image itself is
213    read, resized, or normalised.
214
215    Args:
216        path: A scalar string tensor, the image file path.
217        label: A scalar integer tensor, passed through unchanged.
218
219    Returns:
220        A tuple (image, label), where image is a (224, 224, 3) float32
221        tensor with pixel values normalised to [0, 1].
222    """
223    image = tf.io.read_file(path)
224    image = tf.image.decode_jpeg(image, channels=3)
225    image = tf.image.resize(image, IMAGE_SIZE)
226    image = image / 255.0  # normalise pixel values to [0, 1]
227    return image, label

230def make_pass_fail_dataset(df: pd.DataFrame, split: str, shuffle: bool = True) -> tf.data.Dataset:
231    """Builds a tf.data.Dataset for the core pass/fail classification task.
232
233    Args:
234        df: The DataFrame returned by build_image_level_labels().
235        split: Which split to build the dataset for ("train", "valid", "test").
236        shuffle: Whether to shuffle the dataset. Should be True for training
237            and False for validation/test, to keep evaluation order stable
238            and reproducible.
239
240    Returns:
241        A batched, prefetched tf.data.Dataset yielding (image, label) pairs,
242        where image is a (224, 224, 3) float32 tensor normalised to [0, 1],
243        and label is an integer: 0 for pass, 1 for fail.
244    """
245    split_df = df[df["split"] == split].copy()
246    split_df["label"] = (split_df["pass_fail"] == "fail").astype(int)
247
248    paths = split_df["image_path"].tolist()
249    labels = split_df["label"].tolist()
250
251    ds = tf.data.Dataset.from_tensor_slices((paths, labels))
252    ds = ds.map(_load_and_preprocess_image, num_parallel_calls=tf.data.AUTOTUNE)
253    if shuffle:
254        ds = ds.shuffle(buffer_size=max(len(paths), 1))
255    ds = ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
256    return ds

259def make_defect_type_dataset(
260    df: pd.DataFrame,
261    split: str,
262    class_names: list[str],
263    shuffle: bool = True,
264) -> tf.data.Dataset:
265    """Builds a tf.data.Dataset for the defect-type extension task.
266
267    Only images with a single, unambiguous defect type are included in this
268    dataset — pass images and the multi-defect images (extension_label is
269    None for both, per build_image_level_labels()) are excluded, per the
270    project's documented labelling policy (Section 2 of the project
271    documentation).
272
273    Args:
274        df: The DataFrame returned by build_image_level_labels().
275        split: Which split to build the dataset for ("train", "valid", "test").
276        class_names: Class ID -> name mapping, from load_class_names(). Used
277            to map each defect-type string label to a fixed integer index,
278            consistent across all three splits.
279        shuffle: Whether to shuffle the dataset. Should be True for training
280            and False for validation/test.
281
282    Returns:
283        A batched, prefetched tf.data.Dataset yielding (image, label) pairs,
284        where image is a (224, 224, 3) float32 tensor normalised to [0, 1],
285        and label is an integer class index into class_names.
286    """
287    split_df = df[(df["split"] == split) & (df["extension_label"].notna())].copy()
288
289    label_to_index = {name: i for i, name in enumerate(class_names)}
290    split_df["label_idx"] = split_df["extension_label"].map(label_to_index)
291
292    paths = split_df["image_path"].tolist()
293    labels = split_df["label_idx"].tolist()
294
295    ds = tf.data.Dataset.from_tensor_slices((paths, labels))
296    ds = ds.map(_load_and_preprocess_image, num_parallel_calls=tf.data.AUTOTUNE)
297    if shuffle:
298        ds = ds.shuffle(buffer_size=max(len(paths), 1))
299    ds = ds.batch(BATCH_SIZE).prefetch(tf.data.AUTOTUNE)
300    return ds

Builds a tf.data.Dataset for the defect-type extension task.

Only images with a single, unambiguous defect type are included in this dataset — pass images and the multi-defect images (extension_label is None for both, per build_image_level_labels()) are excluded, per the project's documented labelling policy (Section 2 of the project documentation).

Arguments:
  • df: The DataFrame returned by build_image_level_labels().
  • split: Which split to build the dataset for ("train", "valid", "test").
  • class_names: Class ID -> name mapping, from load_class_names(). Used to map each defect-type string label to a fixed integer index, consistent across all three splits.
  • shuffle: Whether to shuffle the dataset. Should be True for training and False for validation/test.
Returns:

A batched, prefetched tf.data.Dataset yielding (image, label) pairs, where image is a (224, 224, 3) float32 tensor normalised to [0, 1], and label is an integer class index into class_names.
